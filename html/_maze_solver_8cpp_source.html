26bool MazeSolver::loadFromFile(const std::string& filename) {
27 std::ifstream file(filename);
28 if (!file.is_open()) return false;
29
30 if (!(file >> rows >> cols)) return false;

32
33 for (int i = 0; i < rows; ++i) {
34 std::string line;

37 grid[i][j] = line[j] - '0';

41 }
42 }
43 file.close();

52bool MazeSolver::explore(int r, int c) {
53 if (grid[r][c] == 3) return true;
54
55 bool isEntrance = (grid[r][c] == 2);
56 if (!isEntrance) grid[r][c] = 4; // Mark path
57
58 // Direction vectors: Up, Down, Left, Right
59 int dr[] = { -1, 1, 0, 0 };
60 int dc[] = { 0, 0, -1, 1 };
61
62 for (int i = 0; i < 4; ++i) {
63 int nr = r + dr[i];
64 int nc = c + dc[i];
65 if (isValid(nr, nc)) {
66 if (explore(nr, nc)) return true;
67 }
68 }
69
70 if (!isEntrance) grid[r][c] = 0; // Backtrack
71 return false;
72}
